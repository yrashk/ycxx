if(YCXXC_STDLIB STREQUAL "system")
  set(YCXXC_PORTABLE_EXECUTION_SOURCE_DIR "${YCXXC_LIBYCXX_SOURCE_DIR}/portable")
  if(NOT EXISTS "${YCXXC_PORTABLE_EXECUTION_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "The system standard library build needs libycxx's portable execution component")
  endif()
  block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS ${YCXXC_BUILD_SHARED_LIBS})
    if(YCXXC_INSTALL)
      set(YCXX_EXECUTION_PORTABLE_INSTALL ON)
    endif()
    add_subdirectory("${YCXXC_PORTABLE_EXECUTION_SOURCE_DIR}" libycxx-execution EXCLUDE_FROM_ALL SYSTEM)
  endblock()
  set(provider system.cppm)
else()
  set(provider libycxx.cppm)
endif()
add_library(ycxxc_execution ${YCXXC_COMPONENT_TYPE})
target_sources(ycxxc_execution PUBLIC FILE_SET CXX_MODULES FILES "${provider}")
target_link_libraries(ycxxc_execution PUBLIC ycxxc_options)
if(YCXXC_STDLIB STREQUAL "system")
  target_compile_options(ycxx_execution_portable_modules PRIVATE -freflection -fcontracts)
  target_link_libraries(ycxxc_execution PUBLIC ycxx::execution_portable_modules)
endif()
