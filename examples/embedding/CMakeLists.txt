cmake_minimum_required(VERSION 3.30)
if(NOT TARGET ycxx::engine)
  project(ycxx_embedding_example LANGUAGES C CXX)
  set(YCXX_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "ycxx source checkout")
  option(YCXX_EMBED_INSTALLED "Use an installed ycxx package" OFF)
  if(YCXX_EMBED_INSTALLED)
    find_package(ycxx CONFIG REQUIRED COMPONENTS engine compiler)
  else()
    add_subdirectory("${YCXX_SOURCE_DIR}" ycxx EXCLUDE_FROM_ALL)
    set(ycxx_RESOURCE_DIR "${YCXX_SOURCE_DIR}/resources")
  endif()
  enable_testing()
endif()

add_executable(embedded_engine engine.cpp)
target_link_libraries(embedded_engine PRIVATE ycxx::engine)
add_test(NAME embedded_engine COMMAND embedded_engine)

add_executable(embedded_compiler compiler.cpp)
target_link_libraries(embedded_compiler PRIVATE ycxx::compiler)
if(NOT ycxx_RESOURCE_DIR)
  set(ycxx_RESOURCE_DIR "${PROJECT_SOURCE_DIR}/resources")
endif()
add_test(NAME embedded_compiler COMMAND embedded_compiler "${ycxx_RESOURCE_DIR}")

if(NOT YCXX_EMBED_INSTALLED)
  set(ycxx_STDLIB "${YCXXC_STDLIB}")
endif()
if(ycxx_STDLIB STREQUAL "system")
  # Build the peer as a regular host dependency with the compiler's standard headers and runtime.
  add_library(embedded_stock_peer SHARED stock_peer.cpp)
  set_target_properties(embedded_stock_peer PROPERTIES CXX_SCAN_FOR_MODULES OFF)
  add_executable(embedded_stock_runtime stock_runtime.cpp)
  target_link_libraries(embedded_stock_runtime PRIVATE ycxx::engine embedded_stock_peer)
  add_test(NAME embedded_stock_runtime COMMAND embedded_stock_runtime)
endif()

get_directory_property(_embed_tests TESTS)
foreach(_t IN LISTS _embed_tests)
  set_property(TEST "${_t}" APPEND PROPERTY LABELS "smoke;embedding")
endforeach()
